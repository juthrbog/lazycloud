func benchCols() []Column {
	return []Column{
		{Title: "T", Width: 2},
		{Title: "Name", Width: 30, Weight: 3},
		{Title: "Size", Width: 10},
		{Title: "Modified", Width: 19},
		{Title: "Class", Width: 12},
	}
}

func makeRows(n int) ([]table.Row, []table.Row) {
	rows := make([]table.Row, n)
	keys := make([]table.Row, n)
	for i := 0; i < n; i++ {
		name := fmt.Sprintf("logs/2026/10/%06d/object-%d.json", (i*7919)%n, i)
		rows[i] = table.Row{"F", name, fmt.Sprintf("%d KB", i%5000), "2026-10-11 12:00:00", "STANDARD"}
		keys[i] = table.Row{"1", name, SortKeyBytes(int64(i % 5000)), "2026-10-11 12:00:00", "STANDARD"}
	}
	return rows, keys
}

func benchSetRows(b *testing.B, n int, sorted bool, filter string) {
	rows, keys := makeRows(n)
	t := NewTable(benchCols(), nil)
	t.SetSize(160, 40)
	if sorted {
		t.Sort(1)
	}
	if filter != "" {
		t.Filter(filter)
	}
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		t.SetRowsWithSortKeys(rows, keys)
	}
}

func BenchmarkSetRows100k(b *testing.B)             { benchSetRows(b, 100000, false, "") }
func BenchmarkSetRows100kSorted(b *testing.B)       { benchSetRows(b, 100000, true, "") }
func BenchmarkSetRows100kSortedFilter(b *testing.B) { benchSetRows(b, 100000, true, "obj") }

// Simulate progressive load: 100 pages of 1000 rows, SetRows with all rows each page.
func BenchmarkProgressiveLoad100k(b *testing.B) {
	rows, keys := makeRows(100000)
	for i := 0; i < b.N; i++ {
		t := NewTable(benchCols(), nil)
		t.SetSize(160, 40)
		t.Sort(1)
		for p := 1000; p <= len(rows); p += 1000 {
			t.SetRowsWithSortKeys(rows[:p], keys[:p])
		}
	}
}

func bigJSON(lines int) string {
	var b strings.Builder
	b.WriteString("{\n")
	for i := 0; i < lines; i++ {
		fmt.Fprintf(&b, "  \"key%d\": {\"name\": \"value-%d\", \"n\": %d, \"ok\": true},\n", i, i, i)
	}
	b.WriteString("  \"end\": 1\n}")
	return b.String()
}

func benchCV(b *testing.B, lines int) {
	cv := NewContentView("x.json", bigJSON(lines), FormatJSON)
	cv.SetEmbedded(true)
	cv.SetSize(100, 40)
	down := tea.KeyPressMsg{Code: 'j', Text: "j"}
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		cv, _ = cv.Update(down)
		_ = cv.View()
	}
}
